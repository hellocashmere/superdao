# @superdao/icons

React icons exported from the Superdao Figma icon library.

All icons are 24px by default, inherit their color from `currentColor`, and
accept standard SVG props.

```tsx
import { SearchIcon } from "@superdao/icons/outline/search"
import { SearchBoldIcon } from "@superdao/icons/bold/search"

<SearchIcon />
<SearchIcon size={16} />
<SearchBoldIcon size="2rem" title="Search" />
```
